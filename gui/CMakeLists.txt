cmake_minimum_required(VERSION 3.18)
project(sapf_experiments LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SDL2 REQUIRED)
find_package(OpenGL REQUIRED)
find_package(nlohmann_json 3.9 REQUIRED)
set(IMGUI vendor/imgui)
add_library(imgui STATIC ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp
  ${IMGUI}/imgui_widgets.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_demo.cpp
  ${IMGUI}/backends/imgui_impl_sdl2.cpp ${IMGUI}/backends/imgui_impl_opengl3.cpp)
target_include_directories(imgui PUBLIC ${IMGUI} ${IMGUI}/backends)
target_link_libraries(imgui PUBLIC SDL2::SDL2 OpenGL::GL ${CMAKE_DL_LIBS})
add_library(experiment_model model.cpp backend.cpp)
target_include_directories(experiment_model PUBLIC .)
target_link_libraries(experiment_model PUBLIC nlohmann_json::nlohmann_json)
target_compile_options(experiment_model PRIVATE -Wall -Wextra -Wpedantic)
add_executable(sapf-experiments main.cpp ui.cpp)
target_link_libraries(sapf-experiments PRIVATE experiment_model imgui)
target_compile_options(sapf-experiments PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(model-tests tests/model_test.cpp)
  target_link_libraries(model-tests PRIVATE experiment_model)
  add_test(NAME expression-history COMMAND model-tests)
endif()
install(TARGETS sapf-experiments RUNTIME DESTINATION bin)
